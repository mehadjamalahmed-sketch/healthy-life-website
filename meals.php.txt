foreach ($types as $type) {
    $count_query = "SELECT COUNT(*) as count FROM meals" . ($type != 'all' ? " WHERE type = '$type'" : "");
    $count_result = mysqli_query($conn, $count_query);
    $counts[$type] = mysqli_fetch_assoc($count_result)['count'];
}
