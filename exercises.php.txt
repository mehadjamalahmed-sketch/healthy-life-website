<?php

    redirect('index.php');

// جلب التمارين من قاعدة البيانات
$filter = isset($_GET['filter']) ? $_GET['filter'] : 'all';
$exercises = getExercises($conn, $filter);

// تعريف أنواع التمارين للعرض

    'cardio' => 'كارديو',
    'strength' => 'قوة',
    'flexibility' => 'مرونة'
];

// جلب عدد التمارين لكل نوع
$counts = [];
$types = ['all', 'cardio', 'strength', 'flexibility'];
foreach ($types as $type) {
    $count_query = "SELECT COUNT(*) as count FROM exercises" . ($type != 'all' ? " WHERE type = '$type'" : "");
    $count_result = mysqli_query($conn, $count_query);
    $counts[$type] = mysqli_fetch_assoc($count_result)['count'];
}
